async function fetchData(isSilent = false) { const refreshIcon = document.getElementById('refreshIcon'); const btnRefresh = document.getElementById('btnRefresh'); // Liga a animação de girar SOMENTE se não for refresh silencioso if (!isSilent && refreshIcon) { refreshIcon.classList.add('animate-spin'); } if (!isSilent && btnRefresh) { btnRefresh.disabled = true; } const savedFilters = { search: document.getElementById('searchInput')?.value || '', atrasada: document.getElementById('atrasadaFilter')?.value || '', apontamento: document.getElementById('apontamentoAbertoFilter')?.value || '', segmento: document.getElementById('segmentoFilter')?.value || '', producao: document.getElementById('producaoFilter')?.value || '', cliente: document.getElementById('clientFilter')?.value || '', engenharia: document.getElementById('engenhariaFilter')?.value || '', compras: document.getElementById('comprasFilter')?.value || '', terceiros: document.getElementById('terceirosFilter')?.value || '' }; try { const cacheBusterUrl = `${API_URL}?_t=${Date.now()}`; const response = await fetch(cacheBusterUrl, { cache: 'no-store' }); if (!response.ok) { throw new Error(`Erro na requisição: ${response.status} ${response.statusText}`); } rawData = await response.json(); initializeCurrentDate(); populateSelectOptions(savedFilters); const searchInput = document.getElementById('searchInput'); if (searchInput) searchInput.value = savedFilters.search; renderView(); } catch (error) { console.error('Erro ao buscar dados da API:', error); if (!isSilent) { alert('Não foi possível atualizar os dados de produção. Verifique a conexão com o servidor.'); } } finally { // DESLIGA A ANIMAÇÃO DE GIRAR OBRIGATORIAMENTE if (refreshIcon) { refreshIcon.classList.remove('animate-spin'); } if (btnRefresh) { btnRefresh.disabled = false; } // Renderiza os ícones estáticos do Lucide sem reativar a animação lucide.createIcons(); } }
